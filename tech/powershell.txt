#get a list of sub-folder names and store in $folders

$folders	#to print

#recover drive space (multi-line):
$disk = Get-WmiObject Win32_logicalDisk -ComputerName <hostname> -Filter "DeviceID='C:'" 
Select-Object Size,Freespace

$size_format = ($disk.Size / 1073741824)
$free_format = ($disk.Freespace / 1073741824)

echo ""
echo "C: Drive" "Total Size: $size_format"
echo "Free Space: $free_format"
$disk_percentage = ($free_format / $size_format) * 100
echo "Percentage: $disk_percentage %"
echo ""

#get a list of processes for any greater than 0.0 usage
